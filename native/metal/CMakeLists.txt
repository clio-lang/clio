set(CMAKE_OBJCXX_COMPILER "${CMAKE_CXX_COMPILER}")
enable_language(OBJCXX)
foreach(VARIANT IN ITEMS debug optimized sanitized thread-sanitized)
  if(VARIANT STREQUAL "sanitized")
    set(RUNTIME_TARGET kalandra-native-runtime-sanitized)
    set(METAL_FLAGS -O1 -g -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  elseif(VARIANT STREQUAL "thread-sanitized")
    set(RUNTIME_TARGET kalandra-native-runtime-thread-sanitized)
    set(METAL_FLAGS -O1 -g -fsanitize=thread -fno-omit-frame-pointer)
  elseif(VARIANT STREQUAL "optimized")
    set(RUNTIME_TARGET kalandra-task-core-optimized)
    set(METAL_FLAGS -O2)
  else()
    set(RUNTIME_TARGET kalandra-native-runtime)
    set(METAL_FLAGS -O0)
  endif()
  add_library(kalandra-metal-${VARIANT} STATIC Metal.mm)
  set_target_properties(kalandra-metal-${VARIANT} PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED ON)
  target_compile_options(kalandra-metal-${VARIANT} PRIVATE -Wall -Wextra -Werror -fobjc-arc -fblocks ${METAL_FLAGS})
  target_link_libraries(kalandra-metal-${VARIANT} PUBLIC ${RUNTIME_TARGET} "-framework Foundation" "-framework Metal")
  if(BUILD_TESTING)
    add_executable(kalandra-metal-tests-${VARIANT} Tests.cpp)
    target_compile_options(kalandra-metal-tests-${VARIANT} PRIVATE -Wall -Wextra -Werror ${METAL_FLAGS})
    target_link_options(kalandra-metal-tests-${VARIANT} PRIVATE ${METAL_FLAGS})
    target_link_libraries(kalandra-metal-tests-${VARIANT} PRIVATE kalandra-metal-${VARIANT})
    add_test(NAME native-metal-${VARIANT} COMMAND kalandra-metal-tests-${VARIANT})
    set_tests_properties(native-metal-${VARIANT} PROPERTIES TIMEOUT 60 RESOURCE_LOCK apple-metal-gpu ENVIRONMENT "MTL_DEBUG_LAYER=1")
  endif()
endforeach()
