cmake_minimum_required(VERSION 3.28)

file(READ "${CMAKE_CURRENT_LIST_DIR}/native/toolchain-proof/toolchain.json" KALANDRA_TOOLCHAIN_LOCK)
string(JSON KALANDRA_LLVM_VERSION GET "${KALANDRA_TOOLCHAIN_LOCK}" llvmVersion)
string(JSON KALANDRA_DEFAULT_LLVM_ROOT GET "${KALANDRA_TOOLCHAIN_LOCK}" defaultLlvmRoot)
if(DEFINED ENV{KALANDRA_LLVM_ROOT})
  set(KALANDRA_DEFAULT_LLVM_ROOT "$ENV{KALANDRA_LLVM_ROOT}")
endif()
set(KALANDRA_LLVM_ROOT "${KALANDRA_DEFAULT_LLVM_ROOT}" CACHE PATH "Pinned LLVM/MLIR installation")
if(NOT DEFINED CMAKE_C_COMPILER)
  set(CMAKE_C_COMPILER "${KALANDRA_LLVM_ROOT}/bin/clang" CACHE FILEPATH "C compiler")
endif()
if(NOT DEFINED CMAKE_CXX_COMPILER)
  set(CMAKE_CXX_COMPILER "${KALANDRA_LLVM_ROOT}/bin/clang++" CACHE FILEPATH "C++ compiler")
endif()
project(KalandraNative LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The first native proof is validated only on macOS ARM64")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL KALANDRA_LLVM_VERSION OR
   NOT CMAKE_C_COMPILER_VERSION VERSION_EQUAL KALANDRA_LLVM_VERSION)
  message(FATAL_ERROR "Use Clang ${KALANDRA_LLVM_VERSION} from KALANDRA_LLVM_ROOT")
endif()
include(CTest)
add_subdirectory(native)
